extends SceneTree
var game: Control
var failures=0
var count=0
func _initialize() -> void: call_deferred("run")
func ck(ok: bool,label: String) -> void:
	count+=1
	if not ok: failures+=1; printerr("FAIL "+label)
func click(id: String) -> void:
	var b=game.buttons[id]
	var pos=b.get_global_rect().get_center()
	var motion=InputEventMouseMotion.new(); motion.position=pos; root.push_input(motion)
	var e=InputEventMouseButton.new(); e.button_index=MOUSE_BUTTON_LEFT; e.position=pos; e.pressed=true; root.push_input(e)
	await process_frame
	e=e.duplicate();e.pressed=false;root.push_input(e)
	await process_frame
func shot(name: String) -> void:
	await process_frame
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://art/earth-production/"+name+".png")
func run() -> void:
	root.size=Vector2i(1600,900)
	game=load("res://scenes/earth/main.tscn").instantiate()
	root.add_child(game)
	game.transient=true
	await process_frame
	await shot("01-title")
	await click("start")
	ck(game.view=="intro","start mouse")
	await shot("02-opening")
	await click("next")
	ck(game.model.s.intro==1,"next mouse")
	await click("history")
	ck(game.view=="history","history mouse")
	await click("back")
	await click("skip")
	ck(game.view=="base" and game.model.s.people.size()==5,"skip to five-person room")
	await shot("03-room")
	await click("depart")
	await click("mission1")
	ck(game.view=="battle" and game.model.active(),"mission deploy mouse")
	game.set_process(false)
	for i in range(260):game.model.tick(1.0/60)
	game.battlefield.queue_redraw()
	game.refresh_battle()
	await shot("04-battle")
	await click("pause")
	ck(game.paused,"pause mouse")
	await click("speed")
	ck(game.battle_speed==2,"speed mouse")
	await click("recall")
	ck(game.view=="result" and not game.model.active(),"recall mouse")
	await click("return")
	var previous=game.model.s.people[0].level
	await click("grow0")
	ck(game.model.s.people[0].level==previous+1,"growth mouse")
	# UI fixture only: unlock the already model-tested menus for pointer coverage.
	game.model.s.credits=100000;game.model.s.alloy=10000;game.model.s.cores=100
	game.model.s.wins={"3":1,"5":1,"6":1};game.model.s.max_stage=7
	game.model.s.support={"laser":{"level":0,"enabled":false}}
	await click("room")
	await click("room_upgrade")
	ck(game.model.s.room==2,"room upgrade mouse")
	await click("back")
	await click("suits")
	await click("person1")
	await click("make2")
	ck(2 in game.model.s.people[1].owned_suits,"make Fenfen suit mouse")
	await click("equip2")
	ck(game.model.s.people[1].suit==2 and game.model.s.people[0].suit==1,"per person bundle equip mouse")
	await shot("05-suits")
	await click("back")
	await click("growth")
	await click("upgrade_equipment")
	await click("upgrade_talent")
	ck(game.model.s.people[1].equipment==2 and game.model.s.people[1].talent==2,"independent upgrades mouse")
	await shot("06-growth")
	await click("back")
	await click("support")
	await click("buy_laser")
	ck(game.model.s.support.laser.level==1 and game.model.s.support.laser.enabled,"support buy mouse")
	await click("toggle_laser")
	ck(not game.model.s.support.laser.enabled,"support toggle mouse")
	await shot("07-support")
	await click("back")
	await click("depart")
	await click("next_page")
	ck(game.mission_page==1 and game.buttons.mission11.disabled,"page and lock mouse")
	await click("previous_page")
	await shot("08-missions")
	await click("back")
	await click("settings")
	await click("large")
	ck(game.settings.large,"large text mouse")
	await click("back")
	ck(game.view=="base","settings returns to room")
	for b in game.buttons.values():ck(Rect2(0,0,1600,900).encloses(b.get_global_rect()),"button on screen: "+b.name)
	print("EARTH POINTER ",count," checks / ",failures," failures")
	quit(1 if failures else 0)
